{{/* Returns the speaker pages of a session, sorted by surname. Expects dict "keys" (the session's speakers: list) and "year". Speaker keys repeat across years (the same person speaks twice), so the speakers of the session's own year are preferred; pages from any year are used only when the year has none. */}} {{ $keys := .keys | default slice }} {{ $found := slice }} {{ with $keys }} {{ $found = where (where site.RegularPages "Section" $.year) "Params.key" "in" . }} {{ if not $found }}{{ $found = where site.RegularPages "Params.key" "in" . }}{{ end }} {{ end }} {{ return sort $found "Params.surname" }}
