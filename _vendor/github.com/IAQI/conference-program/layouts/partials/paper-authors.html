{{/* Renders a paper's author list, emphasising the presenting author(s). Presenters come from data/presenters-YYYY.yml, generated by scripts/mark_presenters.py (paper id -> list of "First Last" names). Years without such a file render every author unemphasised. Expects a dict with: paper - the accepted-paper record currentYear - the conference year affiliations - print each author's affiliation (optional) presenters - emphasise the presenting author(s) (optional, default true) emphasis - "strong" (default) or "em". Schedule cards use italics, bold reads as too heavy there. (optional) */}} {{ $paper := .paper }} {{ $affiliations := .affiliations }} {{ $italic := eq (.emphasis | default "strong") "em" }} {{ $presenters := slice }} {{ if ne .presenters false }} {{ $byYear := index site.Data (printf "presenters-%s" (string .currentYear)) | default dict }} {{ $presenters = index $byYear (printf "%d" (int $paper.pid)) | default slice }} {{ end }} {{ $authors := $paper.authors }}
{{- range $key, $author := $authors }} {{- $name := trim (printf "%s %s" ($author.first | default "") ($author.last | default "")) " " }} {{- if in $presenters $name -}} {{ if $italic }}{{ $name }}{{ else }}{{ $name }}{{ end }} {{- else }}{{ $name }}{{ end }} {{- if and $affiliations $author.affiliation }} ({{ $author.affiliation }}){{ end }} {{- if ne $key (sub (len $authors) 1) }}; {{ end }} {{- end }}
